cmake_minimum_required(VERSION 3.20)
project(ods_d1_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "@loader_path")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT d05706b4f32a545814856e98ca518292585443b9 CACHE STRING "" FORCE)
set(LLAMA_BUILD_NUMBER 0 CACHE STRING "" FORCE)
if(NOT LLAMA_SOURCE_DIR)
 message(FATAL_ERROR "LLAMA_SOURCE_DIR is required")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(ods_d1 SHARED bridge.cpp)
target_include_directories(ods_d1 PRIVATE "${LLAMA_SOURCE_DIR}/tools/server" "${LLAMA_SOURCE_DIR}/tools/mtmd" "${LLAMA_SOURCE_DIR}/common" "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include")
target_link_libraries(ods_d1 PRIVATE server-context)
set_target_properties(ods_d1 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(ods_d1_error_test EXCLUDE_FROM_ALL bridge_error_test.cpp)
target_include_directories(ods_d1_error_test PRIVATE "${LLAMA_SOURCE_DIR}/tools/server" "${LLAMA_SOURCE_DIR}/tools/mtmd" "${LLAMA_SOURCE_DIR}/common" "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include")
target_link_libraries(ods_d1_error_test PRIVATE server-context)
set_target_properties(ods_d1_error_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
